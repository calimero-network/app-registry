# Everything public is crawlable. The paths below are not secrets — they are
# behind a session and render an empty shell to a crawler, so listing them
# keeps that shell out of the index rather than protecting anything.

User-agent: *
Allow: /
Disallow: /admin
Disallow: /login
Disallow: /my-packages
Disallow: /api/

# ⚠️ ANCHORED, OR IT TAKES THE ORGANISATION PAGES WITH IT. `/orgs` is "my
# organizations" and is behind a session; `/orgs/<id>` is a PUBLIC profile
# listing what that organisation publishes, and a bare `Disallow: /orgs`
# matches by prefix, so it would have hidden every one of them.
Disallow: /orgs$
